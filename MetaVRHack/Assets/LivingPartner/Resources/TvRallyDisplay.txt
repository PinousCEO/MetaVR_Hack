<script>
'use strict';
const el = id => document.getElementById(id);
const base = location.pathname.replace(/\/$/,'');
const canvas = el('court'), ctx = canvas.getContext('2d'), nilo = new Image();
nilo.src = base + '/nilo.png';
let state = {phase:'Join',side:-1,score:0,elapsed:0,marks:0,notice:''}, connected = false, stamp = performance.now(), lastGood = 0;
const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
const phases = {
 Join:['Stand beside the TV','Stand one step back with your partner and the TV in view. Pinch with your right hand or select Play together.'],
 Align:['Locate the court','Look at the center of the TV and pinch with your right hand.'],
 Serve:['Nilo serves','Lower your arms and watch for the ball.'],
 Reach:['','Reach toward the ball, then lower your arm. Keep your partner in the headset view.'],
 Volley:['Quest player’s turn','Return the incoming ball to Nilo with your hand.'],
 Return:['A point for both players!','Lower your arms for the next serve.'],
 Miss:['Try the next one','Lower your arms for another serve.'],
 Won:['Six rallies!','You returned six balls to Nilo together.']
};
function write(id, str) { if (el(id).textContent !== str) el(id).textContent = str; }
function layout() {
 const s=state, words=(phases[s.phase]||phases.Join).slice();
 if(s.phase==='Reach') words[0]=s.side<0?'Reach left!':'Reach right!';
 if(s.phase==='Align' && s.marks===1) words[1]='Stand beside the TV with arms down. Look at your partner’s chest and pinch again.';
 if(s.phase==='Align' && s.marks===2) words[1]='Hold still with both shoulders in the headset view.';
 const notice=s.notice||'';
 if(notice.startsWith('Allow camera')) words[1]='Allow camera access for Yonder in Quest app permissions.';
 if(notice.startsWith('Enable camera')) words[1]='Enable the camera in the headset bag settings.';
 if(notice.startsWith('Waiting for color')) words[1]='Wear the headset and enable passthrough.';
 if(notice.startsWith('Body play')) words[1]='Open the game on Quest 3 or 3S for camera-based play.';
 if(notice.indexOf('steadily')>=0) words[1]='Keep your partner’s shoulders inside the rings.';
 if(/[가-힣]/.test(notice)) words[1]=notice.replace(/\s+/g,' ').trim();
 if(!connected){ words[0]='Waiting for Quest'; words[1]='Open TV Rally in your headset on the same Wi-Fi.'; }
 write('cue',words[0]);write('help',words[1]);write('score',String(s.score));
 write('ready',s.phase==='Won'?'Play again':s.phase==='Join'?'Play together':'Realign court');
 el('ready').disabled=!connected;
}
async function request(path, options){
 const controller=new AbortController(), deadline=setTimeout(()=>controller.abort(),2200);
 try{const response=await fetch(base+path,{...options,cache:'no-store',signal:controller.signal});
 if(!response.ok)throw Error(response.status); return await response.json();}finally{clearTimeout(deadline);}
}
async function poll() {
 try {state=await request('/state'); connected=true; stamp=lastGood=performance.now();}
 catch(e){if(performance.now()-lastGood>2500) connected=false;}
 layout();setTimeout(poll,180);
}
el('ready').onclick=async()=>{el('ready').disabled=true;try{await request('/ready',{method:'POST'});}catch(e){connected=false;layout();}};
el('full').onclick=async()=>{try{
 if(document.fullscreenElement||document.webkitFullscreenElement)await (document.exitFullscreen?document.exitFullscreen():document.webkitExitFullscreen());
 else if(document.documentElement.requestFullscreen)await document.documentElement.requestFullscreen();else document.documentElement.webkitRequestFullscreen();
 }catch(e){write('help','Use your browser’s View menu to enter full screen.');}};
for(const event of ['fullscreenchange','webkitfullscreenchange']) document.addEventListener(event,()=>write('full',(document.fullscreenElement||document.webkitFullscreenElement)?'Exit full screen':'Full screen'));
function render(now){
 const w=canvas.clientWidth,h=canvas.clientHeight,dpr=Math.min(2,devicePixelRatio||1);
 if(canvas.width!==Math.round(w*dpr)||canvas.height!==Math.round(h*dpr)){canvas.width=Math.round(w*dpr);canvas.height=Math.round(h*dpr);}
 ctx.setTransform(dpr,0,0,dpr,0,0);
 const theme=ctx.createLinearGradient(0,0,0,h);theme.addColorStop(0,'#343577');theme.addColorStop(.62,'#4f509a');theme.addColorStop(1,'#686db6');
 ctx.fillStyle=theme;ctx.fillRect(0,0,w,h);
 const floor=h*.68;
 ctx.strokeStyle='rgba(245,245,255,.24)';ctx.lineWidth=Math.max(1,w*.0018);
 for(let i=-3;i<=3;i++){ctx.beginPath();ctx.moveTo(w*.5+i*w*.12,floor);ctx.lineTo(w*.5+i*w*.36,h);ctx.stroke();}
 for(const y of [floor,h*.76,h*.9]){ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(w,y);ctx.stroke();}
 const mid=w*.5, gateY=h*.48,r=Math.min(w*.104,h*.2);
 const gateX=side=>mid+side*w*.33;
 const phase=state.phase, active=connected && (phase==='Serve'||phase==='Reach');
 for(const side of [-1,1]){
   const x=gateX(side),isLit=active&&side===state.side;
   ctx.save();ctx.globalAlpha=active&&!isLit?.26:.85;
   ctx.strokeStyle=side<0?'#ffbe44':'#68e7e5';ctx.lineWidth=Math.max(4,r*(isLit?.12:.052));
   ctx.beginPath();ctx.arc(x,gateY,r,0,Math.PI*2);ctx.stroke();
   if(isLit){ctx.fillStyle=ctx.strokeStyle;ctx.beginPath();ctx.moveTo(x+side*r*.46,gateY);ctx.lineTo(x-side*r*.2,gateY-r*.36);ctx.lineTo(x-side*r*.2,gateY+r*.36);ctx.fill();}
   ctx.restore();
 }
 const t=state.elapsed+Math.min(.35,(now-stamp)/1000);
 const jump=(phase==='Serve'||phase==='Return')&&!reduce?Math.sin(Math.PI*Math.min(1,t))*.045*h:0;
 ctx.fillStyle='rgba(29,27,65,.37)';ctx.beginPath();ctx.ellipse(mid,h*.82,w*.085,h*.022,0,0,Math.PI*2);ctx.fill();
 if(nilo.complete&&nilo.naturalWidth){
   const bh=Math.min(h*.71,w*.34),bw=bh*nilo.naturalWidth/nilo.naturalHeight;
   ctx.drawImage(nilo,mid-bw/2,h*.83-bh-jump,bw,bh);
 }
 let x=mid+w*.055,y=h*.58,visible=connected&&phase!=='Align'&&phase!=='Join';
 const side=state.side, cap=(n)=>Math.max(0,Math.min(1,n));
 if(phase==='Serve'){const u=cap(t);x+=(gateX(side)-x)*u;y+=(gateY-y)*u-Math.sin(u*Math.PI)*h*.22;}
 else if(phase==='Reach'){x=gateX(side);y=gateY;}
 else if(phase==='Volley'){const u=cap(t/.55);x=gateX(side);y=gateY+(h-gateY)*u;visible=u<.99;}
 else if(phase==='Return'){const u=cap(t/.85);x=gateX(side)+(x-gateX(side))*u;y=h+(h*.58-h)*u-Math.sin(Math.PI*u)*h*.22;}
 else visible=false;
 if(visible){const br=Math.max(5,Math.min(w,h)*.036);ctx.fillStyle='#fff6ba';ctx.strokeStyle='#f4ac2b';ctx.lineWidth=br*.19;ctx.beginPath();ctx.arc(x,y,br,0,Math.PI*2);ctx.fill();ctx.stroke();}
 requestAnimationFrame(render);
}
layout();poll();requestAnimationFrame(render);
